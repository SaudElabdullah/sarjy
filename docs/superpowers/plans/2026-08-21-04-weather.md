# Phase 4 — Weather Context Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Sarjy answers weather questions with real data from Open-Meteo (OpenWeatherMap as optional fallback), never guesses, handles unknown/ambiguous locations and outages gracefully, and every spoken number is traceable to the tool payload.

**Architecture:** The Weather context is stateless: pure value objects (`Location`, `Forecast`, `Condition`) in the domain, a `GetWeather` use case in the application layer that orchestrates geocode → cache → forecast with timeouts and fallback behind `WeatherProvider`/`WeatherCache` ports, and HTTP adapters in infrastructure. A `GetWeatherTool` adapts the use case to the Conversation context's `ToolPort`. A tiny regex intent detector forces a `get_weather` call on weather-shaped questions (PRD W-8) via a new `force_tool_when` hook in `RunTurn`.

**Tech Stack:** httpx (async), respx (HTTP mocking in tests), asyncpg (cache table), pydantic (tool arg validation).

**Spec:** `PRD.md` §7.4 (W-1…W-9), §9.3 `get_weather` declaration. Master plan §2 (Open-Meteo / OWM keys), §3.2 layout.

## Global Constraints

- Provider timeout 2.5 s per call; fallback only on `ProviderUnavailable` (W-5).
- Cache key `"{lat:.2f}:{lon:.2f}:{day}:{units}"`, TTL 600 s (W-4).
- Weather answers require a `get_weather` call in the same turn (W-8).
- Tool results carry `grounding_numbers` for the Phase 5 output guard (W-7).
- Forecast horizon ≤ 7 days.
- Open-Meteo needs no key; OWM adapter is constructed only if `OWM_API_KEY` is set.
- Domain layer imports only stdlib + `sarjy.shared`.

```python
# src/sarjy/contexts/weather/domain/forecast.py
from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from typing import Any

from sarjy.contexts.weather.domain.condition import Condition
from sarjy.contexts.weather.domain.location import Location
from sarjy.contexts.weather.domain.units import Units


def c_to_f(c: float) -> float:
    return round(c * 9 / 5 + 32, 1)


def kph_to_mph(k: float) -> float:
    return round(k / 1.609344, 1)


@dataclass(frozen=True, slots=True)
class Forecast:
    temp_c: float
    temp_f: float
    feels_like_c: float
    condition: Condition
    precip_prob: int
    wind_kph: float
    humidity: int
    high_c: float
    low_c: float
    observed_at: datetime
    source: str
    fetched_at: datetime

    @classmethod
    def from_metric(
        cls, *, temp_c: float, feels_like_c: float, condition: Condition, precip_prob: int, wind_kph: float,
        humidity: int, high_c: float, low_c: float, observed_at: datetime, source: str, fetched_at: datetime,
    ) -> Forecast:
        return cls(
            temp_c=round(temp_c, 1), temp_f=c_to_f(temp_c), feels_like_c=round(feels_like_c, 1), condition=condition,
            precip_prob=int(precip_prob), wind_kph=round(wind_kph, 1), humidity=int(humidity), high_c=round(high_c, 1),
            low_c=round(low_c, 1), observed_at=observed_at, source=source, fetched_at=fetched_at,
        )

    def grounding_numbers(self) -> tuple[float, ...]:
        """Every number Sarjy may legitimately speak, in both unit systems, raw and rounded."""
        base = [
            self.temp_c, self.temp_f, self.feels_like_c, c_to_f(self.feels_like_c), float(self.precip_prob),
            self.wind_kph, kph_to_mph(self.wind_kph), float(self.humidity), self.high_c, c_to_f(self.high_c),
            self.low_c, c_to_f(self.low_c),
        ]
        out: set[float] = set()
        for n in base:
            out.add(float(n))
            out.add(float(round(n)))
        return tuple(sorted(out))

    def to_tool_payload(self, location: Location, units: Units) -> dict[str, Any]:
        return {
            "location": {"name": location.name, "country": location.country, "lat": location.lat, "lon": location.lon},
            "observed_at": self.observed_at.isoformat(),
            "units": units.value,
            "temp_c": self.temp_c, "temp_f": self.temp_f,
            "feels_like_c": self.feels_like_c, "feels_like_f": c_to_f(self.feels_like_c),
            "condition_code": self.condition.code, "condition_text": self.condition.text,
            "precip_prob": self.precip_prob,
            "wind_kph": self.wind_kph, "wind_mph": kph_to_mph(self.wind_kph),
            "humidity": self.humidity,
            "high_c": self.high_c, "high_f": c_to_f(self.high_c),
            "low_c": self.low_c, "low_f": c_to_f(self.low_c),
            "source": self.source, "fetched_at": self.fetched_at.isoformat(),
        }
```

```python
from __future__ import annotations

from datetime import date, timedelta

MAX_DAYS_AHEAD = 7


def parse_when(when: str, today: date) -> date | None:
    w = (when or "now").strip().lower()
    if w in ("now", "today", ""):
        return today
    if w == "tomorrow":
        return today + timedelta(days=1)
    try:
        d = date.fromisoformat(w)
    except ValueError:
        return None
    if d < today or (d - today).days > MAX_DAYS_AHEAD:
        return None
    return d
```

```python
# src/sarjy/contexts/weather/application/get_weather.py
from __future__ import annotations

import asyncio
from dataclasses import dataclass
from datetime import date
from typing import Literal

from sarjy.contexts.weather.application.ports import ProviderUnavailable, WeatherCache, WeatherProvider
from sarjy.contexts.weather.domain.forecast import Forecast
from sarjy.contexts.weather.domain.location import Location
from sarjy.contexts.weather.domain.units import Units
from sarjy.contexts.weather.domain.when import parse_when
from sarjy.observability.logging import get_logger
from sarjy.shared.clock import Clock

log = get_logger(__name__)


@dataclass(frozen=True, slots=True)
class WeatherSuccess:
    forecast: Forecast
    location: Location
    units: Units
    day: date


@dataclass(frozen=True, slots=True)
class WeatherNotFound:
    query: str
    reason: Literal["not_found", "no_home_city", "bad_date"]


@dataclass(frozen=True, slots=True)
class WeatherAmbiguous:
    query: str
    candidates: list[Location]


@dataclass(frozen=True, slots=True)
class WeatherUnavailable:
    reason: str


WeatherResult = WeatherSuccess | WeatherNotFound | WeatherAmbiguous | WeatherUnavailable


def cache_key(loc: Location, day: date, units: Units) -> str:
    return f"{loc.lat:.2f}:{loc.lon:.2f}:{day.isoformat()}:{units.value}"


def _is_ambiguous(cands: list[Location]) -> bool:
    if len(cands) < 2:
        return False
    first = cands[0].name.lower()
    same_name = [c for c in cands if c.name.lower() == first]
    return len({c.country for c in same_name}) >= 2


class GetWeather:
    def __init__(
        self, provider: WeatherProvider, fallback: WeatherProvider | None, cache: WeatherCache, clock: Clock,
        timeout_s: float = 2.5, ttl_s: int = 600,
    ) -> None:
        self.provider, self.fallback, self.cache, self.clock = provider, fallback, cache, clock
        self.timeout_s, self.ttl_s = timeout_s, ttl_s

    async def execute(self, query: str, when: str, units: Units, home_city: str | None) -> WeatherResult:
        q = (query or "").strip()
        if q.lower() in ("here", "my city", "home", ""):
            if not home_city:
                return WeatherNotFound(query=q, reason="no_home_city")
            q = home_city
        day = parse_when(when, self.clock.now().date())
        if day is None:
            return WeatherNotFound(query=q, reason="bad_date")

        explicit_country = "," in q
        try:
            cands = await self._with_fallback(lambda p: p.geocode(q))
        except ProviderUnavailable as e:
            return WeatherUnavailable(reason=str(e) or "unavailable")
        if not cands:
            return WeatherNotFound(query=q, reason="not_found")
        if not explicit_country and _is_ambiguous(cands):
            return WeatherAmbiguous(query=q, candidates=cands[:3])
        loc = cands[0]

        key = cache_key(loc, day, units)
        cached = await self.cache.get(key)
        if cached is not None:
            return WeatherSuccess(forecast=cached, location=loc, units=units, day=day)
        try:
            fc = await self._with_fallback(lambda p: p.forecast(loc, day))
        except ProviderUnavailable as e:
            return WeatherUnavailable(reason=str(e) or "unavailable")
        await self.cache.set(key, fc, self.ttl_s)
        return WeatherSuccess(forecast=fc, location=loc, units=units, day=day)

    async def _with_fallback(self, call):  # type: ignore[no-untyped-def]
        providers = [self.provider] + ([self.fallback] if self.fallback else [])
        last: Exception | None = None
        for p in providers:
            try:
                return await asyncio.wait_for(call(p), timeout=self.timeout_s)
            except TimeoutError:

                log.warning("weather_provider_unavailable", provider=p.name, error=str(e))
                last = e
        raise last or ProviderUnavailable("unavailable")
```

```python
from __future__ import annotations

from datetime import date, datetime

import httpx

from sarjy.contexts.weather.application.ports import ProviderUnavailable
from sarjy.contexts.weather.domain.condition import Condition
from sarjy.contexts.weather.domain.forecast import Forecast
from sarjy.contexts.weather.domain.location import Location
from sarjy.shared.clock import Clock

GEO_URL = "https://geocoding-api.open-meteo.com/v1/search"
FC_URL = "https://api.open-meteo.com/v1/forecast"
CURRENT = "temperature_2m,apparent_temperature,relative_humidity_2m,weather_code,wind_speed_10m,precipitation_probability"
DAILY = "temperature_2m_max,temperature_2m_min,weather_code,precipitation_probability_max"


class OpenMeteoProvider:
    name = "open-meteo"

    def __init__(self, client: httpx.AsyncClient, clock: Clock) -> None:
        self.client, self.clock = client, clock

    async def _get(self, url: str, params: dict[str, str | int | float]) -> dict:  # type: ignore[type-arg]
        try:
            r = await self.client.get(url, params=params, timeout=2.5)
        except httpx.HTTPError as e:
            raise ProviderUnavailable(f"open-meteo network: {e.__class__.__name__}") from e
        if r.status_code >= 500 or r.status_code == 429:
            raise ProviderUnavailable(f"open-meteo http {r.status_code}")
        if r.status_code >= 400:
            raise ProviderUnavailable(f"open-meteo bad request {r.status_code}")
        return r.json()  # type: ignore[no-any-return]

    async def geocode(self, query: str) -> list[Location]:
        data = await self._get(GEO_URL, {"name": query, "count": 5, "language": "en", "format": "json"})
        out: list[Location] = []
        for res in data.get("results", []) or []:
            out.append(Location(
                name=str(res.get("name", "")), country=str(res.get("country") or res.get("country_code") or ""),
                lat=float(res["latitude"]), lon=float(res["longitude"]), admin1=res.get("admin1"),
            ))
        return out[:3]

    async def forecast(self, loc: Location, day: date) -> Forecast:
        data = await self._get(FC_URL, {
            "latitude": loc.lat, "longitude": loc.lon, "current": CURRENT, "daily": DAILY, "timezone": "auto", "forecast_days": 7,
        })
        daily = data.get("daily") or {}
        days: list[str] = daily.get("time") or []
        idx = days.index(day.isoformat()) if day.isoformat() in days else 0
        high = float(daily["temperature_2m_max"][idx])
        low = float(daily["temperature_2m_min"][idx])
        d_code = int(daily["weather_code"][idx])
        d_pop = int(daily["precipitation_probability_max"][idx] or 0)
        now = self.clock.now()
        cur = data.get("current") or {}
        if idx == 0 and cur:
            observed = datetime.fromisoformat(cur["time"]).replace(tzinfo=now.tzinfo)
            return Forecast.from_metric(
                temp_c=float(cur["temperature_2m"]), feels_like_c=float(cur["apparent_temperature"]),
                condition=Condition.from_wmo(int(cur["weather_code"])), precip_prob=int(cur.get("precipitation_probability") or d_pop),
                wind_kph=float(cur["wind_speed_10m"]), humidity=int(cur["relative_humidity_2m"]), high_c=high, low_c=low,
                observed_at=observed, source=self.name, fetched_at=now,
            )
        mid = round((high + low) / 2, 1)
        return Forecast.from_metric(
            temp_c=mid, feels_like_c=mid, condition=Condition.from_wmo(d_code), precip_prob=d_pop, wind_kph=0.0,
            humidity=0, high_c=high, low_c=low, observed_at=datetime.combine(day, datetime.min.time(), tzinfo=now.tzinfo),
            source=self.name, fetched_at=now,
        )
```

```python
from __future__ import annotations

from datetime import UTC, date, datetime

import httpx

from sarjy.contexts.weather.application.ports import ProviderUnavailable
from sarjy.contexts.weather.domain.condition import Condition
from sarjy.contexts.weather.domain.forecast import Forecast
from sarjy.contexts.weather.domain.location import Location
from sarjy.shared.clock import Clock

GEO_URL = "https://api.openweathermap.org/geo/1.0/direct"
ONECALL_URL = "https://api.openweathermap.org/data/3.0/onecall"

# OWM condition id → approximate WMO code so Condition text stays consistent
_OWM_TO_WMO = {2: 95, 3: 53, 5: 63, 6: 73, 7: 45, 800: 0, 801: 1, 802: 2, 803: 3, 804: 3}


def _wmo(owm_id: int) -> int:
    if owm_id in _OWM_TO_WMO:
        return _OWM_TO_WMO[owm_id]
    return _OWM_TO_WMO.get(owm_id // 100, 3)


class OwmProvider:
    name = "owm"

    def __init__(self, client: httpx.AsyncClient, api_key: str, clock: Clock) -> None:
        self.client, self.key, self.clock = client, api_key, clock

    async def _get(self, url: str, params: dict[str, str | int | float]):  # type: ignore[no-untyped-def]
        try:
            r = await self.client.get(url, params={**params, "appid": self.key}, timeout=2.5)
        except httpx.HTTPError as e:
            raise ProviderUnavailable(f"owm network: {e.__class__.__name__}") from e
        if r.status_code >= 400:
            raise ProviderUnavailable(f"owm http {r.status_code}")
        return r.json()

    async def geocode(self, query: str) -> list[Location]:
        data = await self._get(GEO_URL, {"q": query, "limit": 5})
        return [Location(name=d["name"], country=d.get("country", ""), lat=float(d["lat"]), lon=float(d["lon"]),
                         admin1=d.get("state")) for d in data][:3]

    async def forecast(self, loc: Location, day: date) -> Forecast:
        data = await self._get(ONECALL_URL, {"lat": loc.lat, "lon": loc.lon, "units": "metric", "exclude": "minutely,hourly,alerts"})
        now = self.clock.now()
        daily = data.get("daily") or []
        idx = next((i for i, d in enumerate(daily) if datetime.fromtimestamp(d["dt"], UTC).date() == day), 0)
        d = daily[idx] if daily else {}
        high, low = float(d.get("temp", {}).get("max", 0)), float(d.get("temp", {}).get("min", 0))
        pop = int(round(float(d.get("pop", 0)) * 100))
        cur = data.get("current") or {}
        if idx == 0 and cur:
            w = (cur.get("weather") or [{}])[0]
            code = Condition(code=_wmo(int(w.get("id", 803))), text=str(w.get("description", "unknown conditions")))
            return Forecast.from_metric(
                temp_c=float(cur["temp"]), feels_like_c=float(cur.get("feels_like", cur["temp"])), condition=code,
                precip_prob=pop, wind_kph=round(float(cur.get("wind_speed", 0)) * 3.6, 1), humidity=int(cur.get("humidity", 0)),
                high_c=high, low_c=low, observed_at=datetime.fromtimestamp(int(cur["dt"]), UTC), source=self.name, fetched_at=now,
            )
        w = (d.get("weather") or [{}])[0]
        mid = round((high + low) / 2, 1)
        return Forecast.from_metric(
            temp_c=mid, feels_like_c=mid, condition=Condition(code=_wmo(int(w.get("id", 803))), text=str(w.get("description", "unknown conditions"))),
            precip_prob=pop, wind_kph=0.0, humidity=0, high_c=high, low_c=low,
            observed_at=datetime.combine(day, datetime.min.time(), tzinfo=UTC), source=self.name, fetched_at=now,
        )
```

```python
# src/sarjy/contexts/weather/infrastructure/pg_cache.py
from __future__ import annotations

import json
from datetime import datetime

from sarjy.contexts.weather.domain.condition import Condition
from sarjy.contexts.weather.domain.forecast import Forecast
from sarjy.infrastructure_shared.db import Database
from sarjy.shared.clock import Clock


class PgWeatherCache:
    def __init__(self, db: Database, clock: Clock) -> None:
        self.db, self.clock = db, clock

    async def get(self, key: str) -> Forecast | None:
        row = await self.db.fetchrow("select payload, fetched_at from weather_cache where cache_key=$1", key)
        if not row:
            return None
        p = json.loads(row["payload"]) if isinstance(row["payload"], str) else row["payload"]
        if (self.clock.now() - row["fetched_at"]).total_seconds() > int(p["_ttl_s"]):
            return None
        return Forecast(
            temp_c=p["temp_c"], temp_f=p["temp_f"], feels_like_c=p["feels_like_c"],
            condition=Condition(code=p["condition_code"], text=p["condition_text"]), precip_prob=p["precip_prob"],
            wind_kph=p["wind_kph"], humidity=p["humidity"], high_c=p["high_c"], low_c=p["low_c"],
            observed_at=datetime.fromisoformat(p["observed_at"]), source=p["source"], fetched_at=datetime.fromisoformat(p["fetched_at"]),
        )

    async def set(self, key: str, forecast: Forecast, ttl_s: int) -> None:
        p = {
            "temp_c": forecast.temp_c, "temp_f": forecast.temp_f, "feels_like_c": forecast.feels_like_c,
            "condition_code": forecast.condition.code, "condition_text": forecast.condition.text,
            "precip_prob": forecast.precip_prob, "wind_kph": forecast.wind_kph, "humidity": forecast.humidity,
            "high_c": forecast.high_c, "low_c": forecast.low_c, "observed_at": forecast.observed_at.isoformat(),
            "source": forecast.source, "fetched_at": forecast.fetched_at.isoformat(), "_ttl_s": ttl_s,
        }
        await self.db.execute(
            """insert into weather_cache (cache_key, payload, fetched_at) values ($1, $2::jsonb, $3)
               on conflict (cache_key) do update set payload = excluded.payload, fetched_at = excluded.fetched_at""",
            key, json.dumps(p), self.clock.now(),
        )
```

```python
from __future__ import annotations

from sarjy.contexts.conversation.application.ports import FactSnapshotPort
from sarjy.contexts.weather.domain.units import Units
from sarjy.shared.ids import UserId

_IMPERIAL_COUNTRIES = {"united states", "us", "usa", "united states of america", "liberia", "myanmar"}


class UnitsResolver:
    def __init__(self, facts: FactSnapshotPort) -> None:
        self.facts = facts

    async def resolve(self, user_id: UserId, explicit: str | None, country: str | None) -> Units:
        if explicit in ("metric", "imperial"):
            return Units(explicit)
        for f in await self.facts.snapshot(user_id):
            if f.key == "units" and f.value.lower() in ("metric", "imperial"):
                return Units(f.value.lower())
        if country and country.strip().lower() in _IMPERIAL_COUNTRIES:
            return Units.IMPERIAL
        return Units.METRIC
```

```python
# src/sarjy/contexts/weather/tool.py
from __future__ import annotations

from typing import Any

from sarjy.contexts.conversation.application.ports import FactSnapshotPort, ToolResult
from sarjy.contexts.weather.application.get_weather import (
    GetWeather, WeatherAmbiguous, WeatherNotFound, WeatherSuccess, WeatherUnavailable,
)
from sarjy.contexts.weather.application.units_resolver import UnitsResolver
from sarjy.contexts.weather.domain.units import Units
from sarjy.shared.ids import UserId

DECLARATION: dict[str, Any] = {
    "name": "get_weather",
    "description": "Get current conditions or a forecast for a real place on Earth. ALWAYS call this before stating any weather information. Never guess weather.",
    "parameters": {
        "type": "object",
        "properties": {
            "location": {"type": "string", "description": "City or 'city, country'. Use 'here' for the user's home city."},
            "when": {"type": "string", "enum": ["now", "today", "tomorrow"], "description": "Or an ISO date within 7 days."},
            "units": {"type": "string", "enum": ["metric", "imperial"]},
        },
        "required": ["location"],
    },
}


class GetWeatherTool:
    name = "get_weather"
    declaration = DECLARATION

    def __init__(self, use_case: GetWeather, facts: FactSnapshotPort, units: UnitsResolver) -> None:
        self.uc, self.facts, self.units = use_case, facts, units

    async def invoke(self, user_id: UserId, args: dict[str, Any]) -> ToolResult:
        query = str(args.get("location", "")).strip()
        when = str(args.get("when") or "now")
        facts = await self.facts.snapshot(user_id)
        home_city = next((f.value for f in facts if f.key == "home_city"), None)
        # first pass with metric to discover country, then re-resolve units (cheap: second call hits cache)
        first = await self.uc.execute(query, when, Units.METRIC, home_city)
        if isinstance(first, WeatherSuccess):
            units = await self.units.resolve(user_id, args.get("units"), first.location.country)
            res = first if units == Units.METRIC else await self.uc.execute(query, when, units, home_city)
        else:
            res = first

        if isinstance(res, WeatherSuccess):
            return ToolResult(ok=True, data=res.forecast.to_tool_payload(res.location, res.units),
                              grounding_numbers=res.forecast.grounding_numbers(),
                              spoken_summary=res.forecast.spoken_summary(res.location, res.units))
        if isinstance(res, WeatherNotFound):
            spoken = {
                "not_found": f"I couldn't find a place called {res.query}. Could you add the country or check the spelling?",
                "no_home_city": "I don't know where you are yet — which city should I check?",
                "bad_date": "I can only look up to seven days ahead.",
            }[res.reason]
            return ToolResult(ok=False, data={"error": "not_found", "reason": res.reason, "query": res.query}, spoken_error=spoken)
        if isinstance(res, WeatherAmbiguous):
            labels = [c.label for c in res.candidates]
            return ToolResult(ok=False, data={"error": "ambiguous", "query": res.query, "candidates": labels},
                              spoken_error=f"Did you mean {' or '.join(labels)}?")
        assert isinstance(res, WeatherUnavailable)
        return ToolResult(ok=False, data={"error": "unavailable", "reason": res.reason},
                          spoken_error="I can't reach the weather service right now. Want to try again in a minute?")
```

```python
# tests/evals/run_weather_evals.py
"""Runs weather.jsonl against RunTurn with the mock provider and real Gemini. Usage: uv run python tests/evals/run_weather_evals.py"""
from __future__ import annotations

import asyncio
import json
import re
import sys
import uuid
from datetime import UTC, datetime
from pathlib import Path

from sarjy.config import Settings
from sarjy.container import Container
from sarjy.contexts.conversation.application.ports import Fact
from sarjy.contexts.conversation.domain.events import SentenceEvent, ToolStatusEvent
from sarjy.contexts.conversation.domain.turn import TurnInput
from sarjy.contexts.conversation.infrastructure.memory_repos import MemMessages, MemSessions
from sarjy.contexts.weather.infrastructure.mock_provider import MockProvider
from sarjy.shared.clock import FakeClock
from sarjy.shared.ids import UserId
from sarjy.shared.text import to_speech

NUM = re.compile(r"-?\d+(?:\.\d+)?")
WORDS_TO_NUM = {to_speech(str(n)): float(n) for n in range(-40, 121)}


class StaticFacts:
    def __init__(self, d: dict[str, str]) -> None:
        self.d = d

    async def snapshot(self, user_id: UserId) -> list[Fact]:
        return [Fact(k, v, "fact") for k, v in self.d.items()]


def spoken_numbers(text: str) -> set[float]:
    found = {float(n) for n in NUM.findall(text)}
    low = text.lower()
    for words, n in WORDS_TO_NUM.items():
        if words and words in low:
            found.add(n)
    return found


async def run_case(case: dict) -> tuple[bool, str]:  # type: ignore[type-arg]
    settings = Settings(weather_provider="mock", app_env="test")  # type: ignore[call-arg]
    clock = FakeClock(datetime(2026, 8, 21, 12, tzinfo=UTC))
    c = Container.build(settings, connect_db=False)
    c.clock = clock
    c.sessions, c.messages = MemSessions(), MemMessages()
    c.facts = StaticFacts(case.get("setup_facts", {}))
    c.tools._tools.clear()
    c.weather_provider = MockProvider(clock, fail=bool(case.get("provider_fail")))
    c.weather_cache = None
    c._wire_weather()
    c.rebuild_run_turn()

    text, tool_called, tool_numbers = "", False, set()
    async for ev in c.run_turn(TurnInput(UserId(uuid.uuid4()), None, str(uuid.uuid4()), case["user"])):
        if isinstance(ev, ToolStatusEvent) and ev.tool == "get_weather" and ev.state == "start":
            tool_called = True
        if isinstance(ev, SentenceEvent):
            text += " " + ev.sentence.text
    for tc in c.messages.tool_calls:  # type: ignore[union-attr]
        res = tc[4]
        tool_numbers |= {float(v) for v in res.values() if isinstance(v, int | float)}
    text = text.strip()

    if tool_called != case["expect_tool_call"]:
        return False, f"tool_call={tool_called} expected {case['expect_tool_call']} :: {text}"
    if case.get("expect_numbers_subset_of_tool_result") and tool_called:
        extra = {n for n in spoken_numbers(text) if not any(abs(n - t) <= 1.0 for t in tool_numbers)}
        if extra:
            return False, f"ungrounded numbers {extra} :: {text}"
    if any_ := case.get("expect_contains_any"):
        if not any(a.lower() in text.lower() for a in any_):
            return False, f"missing any of {any_} :: {text}"
    for bad in case.get("expect_not_contains", []):
        if bad.lower() in text.lower():
            return False, f"contains forbidden {bad!r} :: {text}"
    return True, text


async def main() -> int:
    cases = [json.loads(l) for l in Path(__file__).with_name("weather.jsonl").read_text().splitlines() if l.strip()]
    results = await asyncio.gather(*(run_case(c) for c in cases))
    fails = [(c["id"], why) for c, (ok, why) in zip(cases, results, strict=True) if not ok]
    for cid, why in fails:
        print(f"FAIL {cid}: {why}")
    print(f"weather evals: {len(cases) - len(fails)}/{len(cases)} passed")
    return 1 if fails else 0


if __name__ == "__main__":
    sys.exit(asyncio.run(main()))
```
