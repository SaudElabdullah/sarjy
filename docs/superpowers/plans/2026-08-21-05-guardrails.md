> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the no-op guards with the seven-layer defence from PRD §7.6: client limits, auth quotas, rule-based input guard, Gemini classifier on uncertain inputs, hardened prompt (already in place), tool-level guards (already in Memory/Weather), stream-aware output guard (prompt-leak, persona-break, numeric grounding), async audit, templated refusals, rate limiting, guard-event logging, and the red-team / benign eval suites that gate CI.

**Architecture:** The Guardrails context is pure domain logic (rules, detectors, templates) + two application services (`InputGuard`, `OutputGuard`) implementing the Conversation ports, + infrastructure (Gemini classifier, Postgres event repo, Postgres rate limiter). Nothing in this context imports from Memory/Weather/Assessment; grounding numbers arrive through `GuardContext.tool_numbers`.

**Tech Stack:** regex, pydantic (classifier schema), google-genai via the existing `LLMPort`, asyncpg.

**Spec:** `PRD.md` §7.6 (G-1…G-15, layer table, threat model, categories), §14.3 eval suites, Appendix A templates.

## Global Constraints

- `GUARD_MODE=enforce` in prod; `shadow` only logs (never in prod).
- Rules decision ≤ 5 ms; classifier path ≤ 400 ms p95, fail-closed on timeout for UNCERTAIN (G-12).
- BLOCK → template, no LLM call (G-2).
- Classifier sees last 4 user turns (G-3).
- 8-gram overlap > 2 with system prompt → cut sentence (G-5).
- Every decision logged to `guardrail_events` (G-10).
- Rate limit 60 / 10 min and 500 / day; anonymous users get half (G-11).
- Red-team ≥ 99 %, benign ≤ 2 % FP (G-15).

```python
# src/sarjy/contexts/guardrails/domain/normalize.py
from __future__ import annotations

import base64
import codecs
import re
import unicodedata

_ZW = re.compile(r"[​-‏ -‮⁠-⁯﻿]")
_CTRL = re.compile(r"[\x00-\x08\x0b\x0c\x0e-\x1f\x7f]")
_LEET = str.maketrans({"4": "a", "3": "e", "1": "i", "0": "o", "5": "s", "7": "t", "@": "a", "$": "s"})
_B64 = re.compile(r"(?<![A-Za-z0-9+/=])([A-Za-z0-9+/]{16,}={0,2})(?![A-Za-z0-9+/=])")
_ALPHA_RUN = re.compile(r"[A-Za-z][A-Za-z' ]{19,}")
_STOP = {"the", "and", "you", "your", "all", "tell", "me", "ignore", "are", "is", "to", "of"}


def _maybe_b64(s: str) -> str | None:
    try:
        raw = base64.b64decode(s, validate=True)
        txt = raw.decode("ascii")
    except Exception:  # noqa: BLE001
        return None
    return txt if txt.isprintable() and any(c.isalpha() for c in txt) else None


def _maybe_rot13(s: str) -> str | None:
    dec = codecs.decode(s, "rot13")
    words = set(re.findall(r"[a-z]+", dec.lower()))
    return dec if len(words & _STOP) >= 2 and len(set(re.findall(r"[a-z]+", s.lower())) & _STOP) < 2 else None


def normalize(text: str) -> str:
    t = unicodedata.normalize("NFKC", text)
    t = _ZW.sub("", t)
    t = _CTRL.sub("", t)
    extras: list[str] = []
    for m in _B64.finditer(t):
        d = _maybe_b64(m.group(1))
        if d:
            extras.append(d)
    for m in _ALPHA_RUN.finditer(t):
        d = _maybe_rot13(m.group(0))
        if d:
            extras.append(d)
    t = t.lower()
    leet = t.translate(_LEET)
    if leet != t and re.search(r"[a-z]", leet):
        extras.append(leet)
    out = " ".join([t, *extras])
    return re.sub(r"\s+", " ", out).strip()
```

```python
from __future__ import annotations

import re
from dataclasses import dataclass
from typing import Literal

from sarjy.contexts.guardrails.domain.categories import SEVERITY, Category
from sarjy.contexts.guardrails.domain.decision import GuardDecision


@dataclass(frozen=True, slots=True)
class Rule:
    id: str
    pattern: re.Pattern[str]
    category: Category
    action: Literal["block", "uncertain"]


def _r(id: str, pat: str, cat: Category, action: Literal["block", "uncertain"] = "block") -> Rule:
    return Rule(id, re.compile(pat, re.I), cat, action)


DEFAULT_RULES: list[Rule] = [

]


class RuleEngine:
    def __init__(self, rules: list[Rule]) -> None:
        self._rules = rules

    def evaluate(self, normalized: str) -> GuardDecision:
        uncertain: Rule | None = None
        for r in self._rules:
            if r.pattern.search(normalized):
                if r.action == "block":
                    return GuardDecision("block", r.category, layer=2, rule_id=r.id, severity=SEVERITY[r.category])
                uncertain = uncertain or r
        if uncertain:
            return GuardDecision("uncertain", uncertain.category, layer=2, rule_id=uncertain.id, severity=SEVERITY[uncertain.category])
        return GuardDecision("allow", layer=2)
```

```python
# src/sarjy/contexts/guardrails/domain/leak_detector.py
from __future__ import annotations

import re
from functools import lru_cache

_WORD = re.compile(r"[a-z0-9']+")


def _ngrams(text: str, n: int) -> set[tuple[str, ...]]:
    w = _WORD.findall(text.lower())
    return {tuple(w[i : i + n]) for i in range(max(0, len(w) - n + 1))}


@lru_cache(maxsize=8)
def _system_grams(system_prompt: str, n: int) -> frozenset[tuple[str, ...]]:
    return frozenset(_ngrams(system_prompt, n))


class LeakDetector:
    def __init__(self, system_prompt: str, n: int = 8, threshold: int = 2) -> None:
        self._grams = _system_grams(system_prompt, n)
        self._n, self._threshold = n, threshold

    def is_leak(self, sentence: str) -> bool:
        return len(_ngrams(sentence, self._n) & self._grams) > self._threshold
```

```python
# src/sarjy/contexts/guardrails/domain/grounding.py
from __future__ import annotations

import re

_UNITS = {"zero": 0, "one": 1, "two": 2, "three": 3, "four": 4, "five": 5, "six": 6, "seven": 7, "eight": 8, "nine": 9,
          "ten": 10, "eleven": 11, "twelve": 12, "thirteen": 13, "fourteen": 14, "fifteen": 15, "sixteen": 16,
          "seventeen": 17, "eighteen": 18, "nineteen": 19}
_TENS = {"twenty": 20, "thirty": 30, "forty": 40, "fifty": 50, "sixty": 60, "seventy": 70, "eighty": 80, "ninety": 90}
_WORDNUM = re.compile(
    r"\b(minus|negative)?\s*((?:(?:one|two|three|four|five|six|seven|eight|nine)\s+hundred(?:\s+and)?\s*)?"
    r"(?:(?:twenty|thirty|forty|fifty|sixty|seventy|eighty|ninety)(?:[-\s](?:one|two|three|four|five|six|seven|eight|nine))?"
    r"|(?:zero|one|two|three|four|five|six|seven|eight|nine|ten|eleven|twelve|thirteen|fourteen|fifteen|sixteen|seventeen|eighteen|nineteen)))"
    r"(?:\s+point\s+((?:zero|one|two|three|four|five|six|seven|eight|nine)(?:\s+(?:zero|one|two|three|four|five|six|seven|eight|nine))*))?\b",
    re.I,
)
_DIGIT = re.compile(r"(?<![\w.])(-?\d+(?:\.\d+)?)(?!\s*(?:am|pm|a\.m\.|p\.m\.|o'clock|st|nd|rd|th)\b)(?![\w.])", re.I)


def _words_to_num(s: str) -> float:
    total, cur = 0, 0
    for w in re.split(r"[-\s]+", s.lower()):
        if w in _UNITS:
            cur += _UNITS[w]
        elif w in _TENS:
            cur += _TENS[w]
        elif w == "hundred":
            cur = (cur or 1) * 100
        elif w == "and":
            continue
    return float(total + cur)


def extract_numbers(sentence: str) -> list[float]:
    out: list[float] = []
    s = sentence
    for m in _WORDNUM.finditer(s):
        val = _words_to_num(m.group(2))
        if m.group(3):
            val += float("0." + "".join(str(_UNITS[w]) for w in m.group(3).lower().split()))
        if m.group(1):
            val = -val
        out.append(val)
    s = _WORDNUM.sub(" ", s)
    s = re.sub(r"\b(minus|negative)\s+(\d)", r"-\2", s, flags=re.I)
    out += [float(m.group(1)) for m in _DIGIT.finditer(s)]
    return out


def ungrounded_numbers(sentence: str, allowed: list[float], tolerance: float = 1.0) -> list[float]:
    return [n for n in extract_numbers(sentence) if not any(abs(n - a) <= tolerance for a in allowed)]
```

```python
# src/sarjy/contexts/guardrails/application/input_guard.py
from __future__ import annotations

import asyncio
from typing import Literal

from sarjy.contexts.guardrails.application.ports import ClassifierPort, GuardEventRepo
from sarjy.contexts.guardrails.domain.categories import SEVERITY
from sarjy.contexts.guardrails.domain.decision import GuardDecision
from sarjy.contexts.guardrails.domain.normalize import normalize
from sarjy.contexts.guardrails.domain.rules import RuleEngine
from sarjy.shared.ids import UserId


class InputGuard:
    def __init__(self, rules: RuleEngine, classifier: ClassifierPort, events: GuardEventRepo,
                 mode: Literal["enforce", "shadow"] = "enforce", classifier_timeout_s: float = 0.4) -> None:
        self.rules, self.clf, self.events, self.mode, self.timeout = rules, classifier, events, mode, classifier_timeout_s

    async def check(self, user_id: UserId, text: str, recent_user_turns: list[str]) -> GuardDecision:
        d = self.rules.evaluate(normalize(text))
        if d.action == "block":
            return await self._finish(user_id, d, kind=f"rule:{d.rule_id}")
        if d.action == "allow":
            return d
        # uncertain → classifier (layer 3)
        turns = [*recent_user_turns[-3:], text] if not recent_user_turns or recent_user_turns[-1] != text else recent_user_turns[-4:]
        try:
            c = await asyncio.wait_for(self.clf.classify(turns), timeout=self.timeout)
        except (TimeoutError, asyncio.TimeoutError):
            blocked = GuardDecision("block", d.category, layer=3, rule_id=d.rule_id, severity=d.severity)
            return await self._finish(user_id, blocked, kind="classifier_timeout")
        if c.is_injection or (c.category and c.confidence >= 0.6):
            cat = "injection" if c.is_injection and not c.category else c.category
            blocked = GuardDecision("block", cat, layer=3, rule_id=d.rule_id, severity=SEVERITY.get(cat or "", 1))
            return await self._finish(user_id, blocked, kind=f"classifier:{cat}", detail={"confidence": c.confidence})
        await self.events.record(user_id=user_id, message_id=None, layer=3, kind="classifier:allow", action="allow_flagged",
                                 severity=0, detail={"rule": d.rule_id, "confidence": c.confidence})
        return GuardDecision("allow", layer=3)

    async def _finish(self, user_id: UserId, d: GuardDecision, kind: str, detail: dict | None = None) -> GuardDecision:  # type: ignore[type-arg]
        action = "block" if self.mode == "enforce" else "shadow_block"
        await self.events.record(user_id=user_id, message_id=None, layer=d.layer, kind=kind, action=action,
                                 severity=d.severity, detail={"category": d.category, **(detail or {})})
        return d if self.mode == "enforce" else GuardDecision("allow", d.category, layer=d.layer, rule_id=d.rule_id)
```

```python
# src/sarjy/contexts/guardrails/application/output_guard.py
from __future__ import annotations

import asyncio
from typing import Literal

from sarjy.contexts.conversation.application.ports import GuardContext, SentenceVerdict
from sarjy.contexts.guardrails.application.ports import GuardEventRepo
from sarjy.contexts.guardrails.domain.grounding import ungrounded_numbers
from sarjy.contexts.guardrails.domain.leak_detector import LeakDetector
from sarjy.contexts.guardrails.domain.persona import is_persona_break
from sarjy.contexts.guardrails.domain.templates import refusal_for


class OutputGuard:
    def __init__(self, events: GuardEventRepo, mode: Literal["enforce", "shadow"] = "enforce") -> None:
        self.events, self.mode = events, mode
        self._pending: set[asyncio.Task[None]] = set()

    def check_sentence(self, sentence: str, ctx: GuardContext) -> SentenceVerdict:
        verdict = self._evaluate(sentence, ctx)
        if verdict.action != "pass":
            self._log(ctx, verdict, sentence)
            if self.mode == "shadow":
                return SentenceVerdict("pass")
            ctx.cut_count += 1
        return verdict

    def _evaluate(self, s: str, ctx: GuardContext) -> SentenceVerdict:
        if LeakDetector(ctx.system_prompt).is_leak(s):
            return SentenceVerdict("replace" if ctx.cut_count == 0 else "cut", refusal_for("prompt_leak"), "prompt_leak")
        if is_persona_break(s):
            return SentenceVerdict("replace" if ctx.cut_count == 0 else "cut", refusal_for("persona_switch"), "persona_break")
        if ctx.tool_numbers and ungrounded_numbers(s, ctx.tool_numbers):
            return SentenceVerdict("cut", None, "ungrounded_number")
        return SentenceVerdict("pass")

    def _log(self, ctx: GuardContext, v: SentenceVerdict, sentence: str) -> None:
        try:
            loop = asyncio.get_running_loop()
        except RuntimeError:
            return
        t = loop.create_task(self.events.record(user_id=ctx.user_id, message_id=None, layer=6, kind=v.kind or "output",
                                                action=v.action, severity=3 if v.kind == "prompt_leak" else 2,
                                                detail={"len": len(sentence)}))
        self._pending.add(t)
        t.add_done_callback(self._pending.discard)
```

```python
# src/sarjy/contexts/guardrails/infrastructure/pg_rate_limiter.py
from __future__ import annotations

from dataclasses import dataclass
from datetime import UTC, datetime, timedelta

from sarjy.infrastructure_shared.db import Database
from sarjy.shared.ids import UserId


@dataclass(frozen=True, slots=True)
class RateLimitResult:
    allowed: bool
    retry_after_s: int = 0


class PgRateLimiter:
    def __init__(self, db: Database, per_10min: int = 60, per_day: int = 500) -> None:
        self.db, self.per_10min, self.per_day = db, per_10min, per_day

    async def hit(self, user_id: UserId, is_anonymous: bool) -> RateLimitResult:
        now = datetime.now(UTC)
        w10 = now.replace(minute=(now.minute // 10) * 10, second=0, microsecond=0)
        wday = now.replace(hour=0, minute=0, second=0, microsecond=0)
        lim10 = self.per_10min // (2 if is_anonymous else 1)
        limday = self.per_day // (2 if is_anonymous else 1)
        q = """insert into rate_limits (user_id, window_start, count) values ($1,$2,1)
               on conflict (user_id, window_start) do update set count = rate_limits.count + 1 returning count"""
        c10 = await self.db.fetchval(q, user_id, w10)
        cday = await self.db.fetchval(q, user_id, wday)
        if c10 > lim10:
            return RateLimitResult(False, int((w10 + timedelta(minutes=10) - now).total_seconds()))
        if cday > limday:
            return RateLimitResult(False, int((wday + timedelta(days=1) - now).total_seconds()))
        return RateLimitResult(True)
```

```python
async def rate_limited(user: CurrentUserDep, request: Request) -> None:
    rl = request.app.state.container.rate_limiter
    if rl is None:
        return
    res = await rl.hit(user.user_id, user.is_anonymous)
    if not res.allowed:
        raise HTTPException(status_code=429, detail="rate_limited", headers={"Retry-After": str(res.retry_after_s)})
```
